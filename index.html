export default async function handler(req, res) { // 設定 CORS 標頭，允許你自己嘅 PWA 跨域調用 res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') { return res.status(200).end(); } const { id } = req.query; if (!id) { return res.status(400).send('Missing video ID'); } // 備用嘅高可用音訊解析源（由 Vercel 海外伺服器去抓，唔受手機本地限制） const candidates = [ `https://inv.tux.pizza/latest_version?id=${id}&itag=140&local=true`, `https://invidious.nerdvpn.de/latest_version?id=${id}&itag=140&local=true`, `https://yt.artemislena.eu/latest_version?id=${id}&itag=140&local=true` ]; for (const streamUrl of candidates) { try { const response = await fetch(streamUrl, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36' } }); if (response.ok && response.body) { // 設定標準音訊串流標頭 res.setHeader('Content-Type', 'audio/mp4'); res.setHeader('Cache-Control', 'public, max-age=3600'); // 將抓到的音訊數據以 Stream 管道直接管接噴回手機 const reader = response.body.getReader(); while (true) { const { done, value } = await reader.read(); if (done) break; res.write(value); } return res.end(); } } catch (e) { console.warn('嘗試下一個串流源...', streamUrl); } } return res.status(502).send('Audio stream currently unavailable'); }
